export default function App() {
  const [screen, setScreen] = useState('home');
  const [category, setCategory] = useState<Category>('Hospital');
  const [city, setCity] = useState('');
  const [business, setBusiness] = useState('');
  const [service, setService] = useState('');
  const [date, setDate] = useState('2026-11-25');
  const [time, setTime] = useState('');
  const [customer, setCustomer] = useState('');
  const [mobile, setMobile] = useState('');
  const [bookings, setBookings] = useState<Booking[]>([]);
  const [selectedBooking, setSelectedBooking] = useState<Booking | null>(null);
  const [selectedDoctor, setSelectedDoctor] = useState<Doctor | null>(null);
  const [doctorSearch, setDoctorSearch] = useState('');

  const categories: Category[] = [
    'Hospital',
    'Gym',
    'Car Service',
    'Salon',
  ];

  const startCategory = (value: Category) => {
    setCategory(value);
    setCity('');
    setBusiness('');
    setService('');
    setScreen('city');
  };

  const createBooking = () => {
    if (!customer.trim() || !mobile.trim() || !time) {
      Alert.alert('Required', 'Please enter your details and select a time.');
      return;
    }

    if (mobile.trim().length < 10) {
      Alert.alert('Invalid mobile number', 'Enter a valid mobile number.');
      return;
    }

    const newBooking: Booking = {
      id: `BF${Date.now().toString().slice(-8)}`,
      category,
      business,
      service,
      date,
      time,
      customer,
      status: 'Pending',
    };

    setBookings(previous => [newBooking, ...previous]);
    setSelectedBooking(newBooking);
    setScreen('success');
  };

  const Header = ({ title, back = true }: {
    title: string;
    back?: boolean;
  }) => (
    <View style={styles.header}>
      {back ? (
        <TouchableOpacity
          accessibilityRole="button"
          accessibilityLabel="Go back to home"
          activeOpacity={0.75}
          style={styles.backButton}
          onPress={() => setScreen('home')}
        >
          <Text style={styles.backButtonIcon}>‹</Text>
          <Text style={styles.backText}>Back</Text>
        </TouchableOpacity>
      ) : <View style={styles.headerSideSpace} />}
      <Text style={styles.headerTitle}>{title}</Text>
      <View style={styles.headerSideSpace} />
    </View>
  );

  const Button = ({ title, onPress }: {
    title: string;
    onPress: () => void;
  }) => (
    <TouchableOpacity
      accessibilityRole="button"
      activeOpacity={0.82}
      style={styles.button}
      onPress={onPress}
    >
      <Text style={styles.buttonText}>{title}</Text>
      <View style={styles.buttonArrowCircle}>
        <Text style={styles.buttonArrow}>›</Text>
      </View>
    </TouchableOpacity>
  );

  const Option = ({ title, onPress }: {
    title: string;
    onPress: () => void;
  }) => (
    <TouchableOpacity
      accessibilityRole="button"
      activeOpacity={0.78}
      style={styles.option}
      onPress={onPress}
    >
      <View style={styles.optionLeadingMark} />
      <Text style={styles.optionText}>{title}</Text>
      <View style={styles.optionArrowCircle}>
        <Text style={styles.arrow}>›</Text>
      </View>
    </TouchableOpacity>
  );

  const renderHome = () => (
    <>
      <View style={styles.homeTopBar}>
        <View>
          <Text style={styles.homeGreeting}>GOOD MORNING 👋</Text>
          <Text style={styles.homeWelcome}>Where would you like to go?</Text>
        </View>
        <TouchableOpacity
          style={styles.notificationButton}
          onPress={() => Alert.alert('Notifications', 'You are all caught up!')}
        >
          <Text style={styles.notificationIcon}>♧</Text>
          <View style={styles.notificationDot} />
        </TouchableOpacity>
      </View>

      <View style={styles.hero}>
        <View style={styles.heroPill}>
          <Text style={styles.heroPillText}>BOOKFLOW  ·  ONE APP, MULTIPLE BOOKINGS</Text>
        </View>
        <Text style={styles.heroTitle}>Your time matters.</Text>
        <Text style={styles.heroSubtitle}>
          Discover trusted services and book appointments without the wait.
        </Text>
        <View style={styles.heroBottom}>
          <View style={styles.heroCheck}><Text style={styles.heroCheckText}>✓</Text></View>
          <Text style={styles.heroBottomText}>Simple booking  ·  Easy updates</Text>
        </View>
        <View style={styles.heroOrbOne} />
        <View style={styles.heroOrbTwo} />
      </View>

      <View style={styles.homeSectionHeader}>
        <View>
          <Text style={styles.sectionTitle}>Explore categories</Text>
          <Text style={styles.subtitle}>Choose a service to get started</Text>
        </View>
        <View style={styles.categoryCount}>
          <Text style={styles.categoryCountText}>04 SERVICES</Text>
        </View>
      </View>

      <View style={styles.grid}>
        {categories.map((item, index) => {
          const icons = ['✚', '⌁', '▰', '✦'];
          const hints = [
            'Doctors & specialists',
            'Fitness & wellness',
            'Vehicle care',
            'Beauty & grooming',
          ];
          return (
            <TouchableOpacity
              key={item}
              accessibilityRole="button"
              accessibilityLabel={`Explore ${item}`}
              activeOpacity={0.78}
              style={styles.categoryCard}
              onPress={() => startCategory(item)}
            >
              <View style={[styles.categoryIconBox, index === 1 && styles.categoryIconBoxMint]}>
                <Text style={styles.categoryIcon}>{icons[index]}</Text>
              </View>
              <Text style={styles.categoryText}>{item}</Text>
              <Text style={styles.categoryDescription}>{hints[index]}</Text>
              <View style={styles.categoryCardFooter}>
                <Text style={styles.categoryHint}>Explore</Text>
                <Text style={styles.categoryArrow}>↗</Text>
              </View>
            </TouchableOpacity>
          );
        })}
      </View>

      <View style={styles.homeSectionHeader}>
        <View>
          <Text style={styles.sectionTitle}>Your BookFlow</Text>
          <Text style={styles.subtitle}>Everything in one place</Text>
        </View>
      </View>
      <TouchableOpacity style={styles.quickAccessCard} activeOpacity={0.85} onPress={() => setScreen('bookings')}>
        <View style={styles.quickAccessIcon}><Text style={styles.quickAccessIconText}>▣</Text></View>
        <View style={styles.quickAccessCopy}>
          <Text style={styles.quickAccessTitle}>My appointments</Text>
          <Text style={styles.quickAccessSubtitle}>{bookings.length} booking request(s) · Track status</Text>
        </View>
        <Text style={styles.quickAccessArrow}>›</Text>
      </TouchableOpacity>
      <TouchableOpacity style={styles.quickAccessCard} activeOpacity={0.85} onPress={() => setScreen('profile')}>
        <View style={[styles.quickAccessIcon, styles.quickAccessIconSoft]}><Text style={styles.quickAccessIconText}>◉</Text></View>
        <View style={styles.quickAccessCopy}>
          <Text style={styles.quickAccessTitle}>My profile</Text>
          <Text style={styles.quickAccessSubtitle}>Manage your contact details</Text>
        </View>
        <Text style={styles.quickAccessArrow}>›</Text>
      </TouchableOpacity>
    </>
  );

  const renderCity = () => (
    <>
      <Header title="Select Your City" />
      <Text style={styles.sectionTitle}>{category}</Text>
      <Text style={styles.subtitle}>
        Choose your preferred location
      </Text>
      {CITIES.map(item => (
        <Option
          key={item}
          title={item}
          onPress={() => {
            setCity(item);
            setScreen('business');
          }}
        />
      ))}
    </>
  );

  const renderBusiness = () => (
    <>
      <Header title="Choose Business" />
      <Text style={styles.sectionTitle}>{category}</Text>
      <Text style={styles.subtitle}>Available in {city}</Text>
      {BUSINESSES[category].map(item => (
        <Option
          key={item}
          title={item}
          onPress={() => {
            setBusiness(item);
            setScreen('service');
          }}
        />
      ))}
    </>
  );

  const renderService = () => (
    <>
      <Header title={category === 'Hospital' ? 'Select Department' : 'Select Service'} />
      <View style={styles.hospitalBanner}>
        <Text style={styles.hospitalBannerTitle}>{business}</Text>
        <Text style={styles.hospitalBannerSubtitle}>{city} · {category}</Text>
      </View>
      <Text style={styles.sectionTitle}>{category === 'Hospital' ? 'Departments & Specialties' : 'Available Services'}</Text>
      <Text style={styles.subtitle}>{category === 'Hospital' ? 'Choose a department to view doctors.' : 'Choose the service you want to book.'}</Text>
      {SERVICES[category].map(item => (
        <Option key={item} title={item} onPress={() => {
          setService(item);
          if (category === 'Hospital') {
            setDoctorSearch('');
            setScreen('doctors');
          } else {
            setScreen('datetime');
          }
        }} />
      ))}
    </>
  );


  const renderDoctors = () => {
    const list = DOCTORS.filter(d =>
      d.department === service &&
      `${d.name} ${d.qualification}`.toLowerCase().includes(doctorSearch.toLowerCase())
    );

    return (
      <>
        <Header title="Choose Your Doctor" />
        <View style={styles.hospitalBanner}>
          <Text style={styles.hospitalBannerTitle}>{business}</Text>
          <Text style={styles.hospitalBannerSubtitle}>{city} · {service} Department</Text>
        </View>
        <Text style={styles.sectionTitle}>Available Specialists</Text>
        <Text style={styles.subtitle}>{list.length} doctor(s) in this department</Text>
        <TextInput
          style={styles.input}
          value={doctorSearch}
          onChangeText={setDoctorSearch}
          placeholder="Search doctor or qualification"
        />
        {list.length === 0 && (
          <View style={styles.doctorEmpty}>
            <Text style={styles.doctorEmptyTitle}>No doctors found</Text>
            <Text style={styles.subtitle}>Try another department or search term.</Text>
          </View>
        )}
        {list.map(doctor => (
          <View key={doctor.id} style={styles.doctorCard}>
            <View style={styles.doctorTop}>
              <View style={styles.doctorAvatar}>
                <Text style={styles.doctorInitials}>{doctor.initials}</Text>
              </View>
              <View style={{ flex: 1 }}>
                <Text style={styles.doctorName}>{doctor.name}</Text>
                <Text style={styles.doctorDepartment}>{doctor.department} Specialist</Text>
                <Text style={styles.doctorQualification}>{doctor.qualification}</Text>
                <Text style={styles.doctorRating}>★ {doctor.rating} · {doctor.reviews} reviews</Text>
              </View>
            </View>
            <View style={styles.doctorDivider} />
            <View style={styles.doctorStats}>
              <View><Text style={styles.statValue}>{doctor.experience} Years</Text><Text style={styles.statLabel}>Experience</Text></View>
              <View><Text style={styles.statValue}>₹{doctor.fee}</Text><Text style={styles.statLabel}>Consultation</Text></View>
              <View><Text style={styles.statValue}>Available</Text><Text style={styles.statLabel}>Sample slots</Text></View>
            </View>
            <TouchableOpacity accessibilityRole="button" activeOpacity={0.82} style={styles.profileButton} onPress={() => {
              setSelectedDoctor(doctor);
              setScreen('doctorProfile');
            }}>
              <Text style={styles.profileButtonText}>View Profile & Book Appointment  ›</Text>
            </TouchableOpacity>
          </View>
        ))}
        <Text style={styles.demoDisclaimer}>Demo doctor profiles and fees are illustrative, not verified provider information.</Text>
      </>
    );
  };

  const renderDoctorProfile = () => {
    if (!selectedDoctor) return null;
    return (
      <>
        <Header title="Doctor Profile" />
        <View style={styles.profileHero}>
          <View style={styles.profileAvatar}>
            <Text style={styles.profileInitials}>{selectedDoctor.initials}</Text>
          </View>
          <Text style={styles.profileDoctorName}>{selectedDoctor.name}</Text>
          <Text style={styles.profileSpeciality}>{selectedDoctor.department} Specialist</Text>
          <Text style={styles.doctorQualification}>{selectedDoctor.qualification}</Text>
          <Text style={styles.doctorRating}>★ {selectedDoctor.rating} · {selectedDoctor.reviews} reviews</Text>
        </View>
        <View style={styles.summary}>
          <Text style={styles.sectionTitle}>Professional Information</Text>
          <Text style={styles.summaryText}>Qualification: {selectedDoctor.qualification}</Text>
          <Text style={styles.summaryText}>Experience: {selectedDoctor.experience} years</Text>
          <Text style={styles.summaryText}>Consultation Fee: ₹{selectedDoctor.fee}</Text>
          <Text style={styles.summaryText}>Department: {selectedDoctor.department}</Text>
        </View>
        <View style={styles.summary}>
          <Text style={styles.sectionTitle}>About Doctor</Text>
          <Text style={styles.summaryText}>{selectedDoctor.about}</Text>
        </View>
        <View style={styles.summary}>
          <Text style={styles.sectionTitle}>Available Working Days</Text>
          {selectedDoctor.availableDays.map(day => (
            <Text key={day} style={styles.summaryText}>✓  {day}</Text>
          ))}
        </View>
        <View style={styles.summary}>
          <Text style={styles.sectionTitle}>Typical Appointment Slots</Text>
          {DOCTOR_TIMES.map(slot => <Text key={slot} style={styles.summaryText}>◷  {slot}</Text>)}
        </View>
        <Button title={`Book Appointment · ₹${selectedDoctor.fee}`} onPress={() => {
          setTime('');
          setScreen('datetime');
        }} />
        <Text style={styles.demoDisclaimer}>Profile and availability are sample demo data. Confirm details with the hospital.</Text>
      </>
    );
  };

  const renderDateTime = () => (
    <>
      <Header title="Select Date & Time" />
      <Text style={styles.sectionTitle}>Appointment Date</Text>
      <TextInput
        style={styles.input}
        value={date}
        onChangeText={setDate}
        placeholder="YYYY-MM-DD"
      />
      {selectedDoctor && <View style={styles.summary}><Text style={styles.summaryText}>Doctor: {selectedDoctor.name}</Text><Text style={styles.summaryText}>Consultation fee: ₹{selectedDoctor.fee}</Text></View>}
      <Text style={styles.sectionTitle}>Available Time Slots</Text>
      <View style={styles.timeGrid}>
        {TIMES.map(item => (
          <TouchableOpacity
            key={item}
            accessibilityRole="button"
            accessibilityState={{ selected: time === item }}
            activeOpacity={0.78}
            style={[
              styles.timeOption,
              time === item && styles.selectedTime,
            ]}
            onPress={() => setTime(item)}
          >
            <Text style={[
              styles.timeText,
              time === item && styles.selectedTimeText,
            ]}>
              {item}
            </Text>
          </TouchableOpacity>
        ))}
      </View>
      <Button
        title="Continue"
        onPress={() => {
          if (!time || !date.trim()) {
            Alert.alert('Select appointment', 'Choose date and time.');
            return;
          }
          setScreen('details');
        }}
      />
    </>
  );

  const renderDetails = () => (
    <>
      <Header title="Customer Details" />
      <Text style={styles.sectionTitle}>Your Information</Text>
      <Text style={styles.label}>Full Name</Text>
      <TextInput
        style={styles.input}
        value={customer}
        onChangeText={setCustomer}
        placeholder="Enter your name"
      />
      <Text style={styles.label}>Mobile Number</Text>
      <TextInput
        style={styles.input}
        value={mobile}
        onChangeText={setMobile}
        placeholder="Enter mobile number"
        keyboardType="phone-pad"
        maxLength={10}
      />
      <View style={styles.summary}>
        <Text style={styles.sectionTitle}>Booking Summary</Text>
        <Text style={styles.summaryText}>Category: {category}</Text>
        <Text style={styles.summaryText}>City: {city}</Text>
        <Text style={styles.summaryText}>Business: {business}</Text>
        <Text style={styles.summaryText}>Service: {service}</Text>
        {selectedDoctor && <Text style={styles.summaryText}>Doctor: {selectedDoctor.name}</Text>}
        {selectedDoctor && <Text style={styles.summaryText}>Consultation Fee: ₹{selectedDoctor.fee}</Text>}
        <Text style={styles.summaryText}>Date: {date}</Text>
        <Text style={styles.summaryText}>Time: {time}</Text>
      </View>
      <Button title="Submit Booking" onPress={createBooking} />
    </>
  );

  const renderSuccess = () => (
    <View style={styles.center}>
      <Text style={{ fontSize: 55 }}>✅</Text>
      <Text style={styles.successTitle}>Booking Submitted!</Text>
      <Text style={styles.subtitle}>
        Your appointment request has been submitted.
      </Text>
      {selectedBooking && (
        <View style={styles.summary}>
          <Text style={styles.summaryText}>
            Booking ID: {selectedBooking.id}
          </Text>
          <Text style={styles.summaryText}>
            Business: {selectedBooking.business}
          </Text>
          <Text style={styles.summaryText}>
            Date: {selectedBooking.date}
          </Text>
          <Text style={styles.summaryText}>
            Time: {selectedBooking.time}
          </Text>
          <Text style={styles.pending}>Status: Pending Approval</Text>
        </View>
      )}
      <Button title="My Bookings" onPress={() => setScreen('bookings')} />
      <TouchableOpacity onPress={() => setScreen('home')}>
        <Text style={styles.link}>Back to Home</Text>
      </TouchableOpacity>
    </View>
  );

  const renderBookings = () => (
    <>
      <Header title="My Bookings" />
      {bookings.length === 0 ? (
        <View style={styles.center}>
          <Text style={{ fontSize: 45 }}>📅</Text>
          <Text style={styles.sectionTitle}>No Bookings Yet</Text>
          <Text style={styles.subtitle}>
            Your appointments will appear here.
          </Text>
          <Button title="Explore Services" onPress={() => setScreen('home')} />
        </View>
      ) : (
        bookings.map(item => (
          <View key={item.id} style={styles.bookingCard}>
            <Text style={styles.bookingTitle}>{item.business}</Text>
            <Text style={styles.summaryText}>{item.category} • {item.service}</Text>
            <Text style={styles.summaryText}>{item.date} | {item.time}</Text>
            <Text style={styles.pending}>● {item.status}</Text>
            {item.status === 'Pending' && (
              <TouchableOpacity
                onPress={() => {
                  Alert.alert(
                    'Demo Admin',
                    'Simulate the admin response?',
                    [
                      { text: 'Keep Pending', style: 'cancel' },
                      {
                        text: 'Approve',
                        onPress: () => {
                          setBookings(previous =>
                            previous.map(booking =>
                              booking.id === item.id
                                ? { ...booking, status: 'Approved' }
                                : booking
                            )
                          );
                        },
                      },
                      {
                        text: 'Reject',
                        onPress: () => {
                          setBookings(previous =>
                            previous.map(booking =>
                              booking.id === item.id
                                ? { ...booking, status: 'Rejected' }
                                : booking
                            )
                          );
                        },
                      },
                    ]
                  );
                }}
              >
                <Text style={styles.link}>Demo: Simulate Admin Response</Text>
              </TouchableOpacity>
            )}
          </View>
        ))
      )}
    </>
  );

  const renderProfile = () => (
    <>
      <Header title="My Profile" />
      <View style={styles.center}>
        <Text style={{ fontSize: 55 }}>👤</Text>
        <Text style={styles.sectionTitle}>BookFlow Customer</Text>
        <Text style={styles.subtitle}>Manage your appointments here.</Text>
      </View>
    </>
  );

  return (
    <SafeAreaProvider>
    <SafeAreaView style={styles.safe} edges={['top', 'left', 'right']}>
      <StatusBar backgroundColor={COLORS.background} barStyle="dark-content" />

      <ScrollView
        style={styles.container}
        contentContainerStyle={{ paddingBottom: 25 }}

          <Text style={[styles.navLabel, screen === 'bookings' && styles.navLabelActive]}>Bookings</Text>
        </TouchableOpacity>

          <Text style={[styles.navIcon, screen === 'profile' && styles.navIconActive]}>♙</Text>
          <Text style={[styles.navLabel, screen === 'profile' && styles.navLabelActive]}>Profile</Text>
        </TouchableOpacity>
      </View>
    </SafeAreaView>
    </SafeAreaProvider>
  );
